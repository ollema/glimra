// Struct definitions
struct VertexInput {
    @location(0) position: vec3<f32>,
    @location(1) normal: vec3<f32>,
    @location(2) tex_coord: vec2<f32>,
}

struct VertexOutput {
    @builtin(position) clip_position: vec4<f32>,
    @location(0) world_normal: vec3<f32>,
    @location(1) world_position: vec3<f32>,
    @location(2) tex_coord: vec2<f32>,
}

struct Camera {
    view_proj: mat4x4<f32>,
    view: mat4x4<f32>,
    position: vec3<f32>,
}

struct Light {
    position: vec3<f32>,
    color: vec3<f32>,
    intensity: f32,
}

struct Material {
    base_color: vec3<f32>,
    metallic: f32,
    roughness: f32,
}

// Uniform bindings
@group(0) @binding(0) var<uniform> camera: Camera;
@group(1) @binding(0) var<uniform> model: mat4x4<f32>;
@group(1) @binding(1) var<uniform> normal_matrix: mat3x3<f32>;
@group(2) @binding(0) var<uniform> light: Light;
@group(2) @binding(1) var<uniform> material: Material;

// Texture bindings
@group(3) @binding(0) var diffuse_texture: texture_2d<f32>;
@group(3) @binding(1) var diffuse_sampler: sampler;
@group(3) @binding(2) var normal_texture: texture_2d<f32>;

// Constants
const PI: f32 = 3.14159265359;
const EPSILON: f32 = 0.0001;

// Vertex shader
@vertex
fn vs_main(in: VertexInput) -> VertexOutput {
    var out: VertexOutput;

    let world_position = model * vec4<f32>(in.position, 1.0);
    out.world_position = world_position.xyz;
    out.world_normal = normalize(normal_matrix * in.normal);
    out.tex_coord = in.tex_coord;
    out.clip_position = camera.view_proj * world_position;

    return out;
}

// Fragment shader
@fragment
fn fs_main(in: VertexOutput) -> @location(0) vec4<f32> {
    // Sample textures
    let albedo = textureSample(diffuse_texture, diffuse_sampler, in.tex_coord).rgb;
    let normal_map = textureSample(normal_texture, diffuse_sampler, in.tex_coord).rgb;

    // Calculate normal from normal map
    let normal = normalize(in.world_normal + (normal_map * 2.0 - 1.0));

    // View direction
    let view_dir = normalize(camera.position - in.world_position);

    // Light direction
    let light_dir = normalize(light.position - in.world_position);

    // Diffuse lighting
    let n_dot_l = max(dot(normal, light_dir), 0.0);
    let diffuse = light.color * n_dot_l * light.intensity;

    // Specular lighting (Blinn-Phong)
    let halfway = normalize(light_dir + view_dir);
    let n_dot_h = max(dot(normal, halfway), 0.0);
    let shininess = (1.0 - material.roughness) * 128.0;
    let specular = light.color * pow(n_dot_h, shininess) * material.metallic;

    // Combine
    let ambient = vec3<f32>(0.03);
    let final_color = (ambient + diffuse) * albedo * material.base_color + specular;

    // Tone mapping and gamma correction
    let mapped = final_color / (final_color + vec3<f32>(1.0));
    let gamma_corrected = pow(mapped, vec3<f32>(1.0 / 2.2));

    return vec4<f32>(gamma_corrected, 1.0);
}

// Compute shader example
struct Particle {
    position: vec3<f32>,
    velocity: vec3<f32>,
    life: f32,
}

@group(0) @binding(0) var<storage, read_write> particles: array<Particle>;
@group(0) @binding(1) var<uniform> delta_time: f32;

@compute @workgroup_size(64)
fn cs_update_particles(@builtin(global_invocation_id) id: vec3<u32>) {
    let index = id.x;
    if (index >= arrayLength(&particles)) {
        return;
    }

    var particle = particles[index];

    // Update position
    particle.position += particle.velocity * delta_time;

    // Apply gravity
    particle.velocity.y -= 9.81 * delta_time;

    // Decrease life
    particle.life -= delta_time;

    // Reset if dead
    if (particle.life <= 0.0) {
        particle.position = vec3<f32>(0.0);
        particle.velocity = vec3<f32>(0.0, 10.0, 0.0);
        particle.life = 5.0;
    }

    particles[index] = particle;
}

// Helper functions
fn calculate_fresnel(cos_theta: f32, f0: vec3<f32>) -> vec3<f32> {
    return f0 + (1.0 - f0) * pow(clamp(1.0 - cos_theta, 0.0, 1.0), 5.0);
}

fn distribution_ggx(n: vec3<f32>, h: vec3<f32>, roughness: f32) -> f32 {
    let a = roughness * roughness;
    let a2 = a * a;
    let n_dot_h = max(dot(n, h), 0.0);
    let n_dot_h2 = n_dot_h * n_dot_h;

    let num = a2;
    var denom = (n_dot_h2 * (a2 - 1.0) + 1.0);
    denom = PI * denom * denom;

    return num / denom;
}
