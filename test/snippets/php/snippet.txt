// Interface
interface Greetable {
    public function greet(): string;
}

// Trait
trait Loggable {
    protected function log(string $message): void {
        echo "[LOG] $message\n";
    }
}

// Enum (PHP 8.1+)
enum Status: string {
    case Active = 'active';
    case Inactive = 'inactive';
    case Pending = 'pending';
}

// Abstract class
abstract class Entity {
    protected int $id;

    abstract public function save(): bool;
}

// Class with traits and interfaces
class Person extends Entity implements Greetable {
    use Loggable;

    private string $name;
    private int $age;
    public readonly DateTime $createdAt;

    public function __construct(
        string $name,
        int $age,
        private Status $status = Status::Active
    ) {
        $this->name = $name;
        $this->age = $age;
        $this->createdAt = new DateTime();
    }

    public function greet(): string {
        return "Hello, I'm {$this->name}";
    }

    public function save(): bool {
        $this->log("Saving person: {$this->name}");
        return true;
    }

    public function getName(): string {
        return $this->name;
    }

    public static function create(string $name, int $age): self {
        return new self($name, $age);
    }
}

// Function with typed parameters

}

// Nullable types
function findUser(?int $id): ?Person {
    if ($id === null) {
        return null;
    }
    return Person::create("User", 25);
}

// Match expression (PHP 8.0+)
function describe(mixed $value): string {
    return match(true) {
        is_int($value) => 'integer',
        is_string($value) => 'string',
        is_array($value) => 'array',
        default => 'unknown',
    };
}
