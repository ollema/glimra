\begin{theorem}[Pythagorean Theorem]
\label{thm:pythagoras}
For a right triangle with legs $a$ and $b$ and hypotenuse $c$:
\[ a^2 + b^2 = c^2 \]
\end{theorem}

def fibonacci(n):
    """Calculate the nth Fibonacci number."""
    if n <= 1:
        return n
    return fibonacci(n-1) + fibonacci(n-2)
\end{lstlisting}
