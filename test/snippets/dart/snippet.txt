// Constants
const int maxSize = 100;
const pi = 3.14159;

// Enum
enum Status { active, inactive, pending }

// Mixin
mixin Loggable {
  void log(String message) {
    print('[LOG] $message');
  }
}

// Abstract class
abstract class Entity {
  int get id;
  bool save();
}

// Class with mixin
class Person extends Entity with Loggable {
  final String name;
  int age;
  Status status;

  // Named constructor
  Person(this.name, this.age, {this.status = Status.active});

  // Factory constructor
  factory Person.guest() {
    return Person('Guest', 0);
  }

  @override
  int get id => name.hashCode;

  @override
  bool save() {
    log('Saving person: $name');
    return true;
  }

  // Getter
  bool get isAdult => age >= 18;

  // Setter
  set statusValue(Status value) => status = value;

  // Method with optional parameters
  String greet({String? greeting}) {
    greeting ??= 'Hello';
    return '$greeting, I\'m $name';
  }

  // Arrow function
  String describe() => '$name is $age years old';
}

// Extension method
extension StringExtension on String {
  String capitalize() {
    if (isEmpty) return this;
    return '${this[0].toUpperCase()}${substring(1)}';
  }
}

// Generic class
class Container<T> {
  final List<T> _items = [];

  void add(T item) => _items.add(item);
  T operator [](int index) => _items[index];
  int get length => _items.length;
}

// Function with type parameters
T? findFirst<T>(List<T> items, bool Function(T) predicate) {
  for (final item in items) {
    if (predicate(item)) return item;
  }
  return null;
}

// Async function
Future<String> fetchData(String url) async {
  await Future.delayed(Duration(seconds: 1));
  return 'Data from $url';
}

// Stream generator
Stream<int> countUp(int max) async* {
  for (var i = 0; i < max; i++) {
    yield i;
  }
}
