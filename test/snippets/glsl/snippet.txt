#version 330 core

// Vertex attributes
layout(location = 0) in vec3 aPosition;
layout(location = 1) in vec3 aNormal;
layout(location = 2) in vec2 aTexCoord;

// Uniforms
uniform mat4 uModel;
uniform mat4 uView;
uniform mat4 uProjection;
uniform mat3 uNormalMatrix;
uniform float uTime;

// Output to fragment shader
out vec3 vNormal;
out vec2 vTexCoord;
out vec3 vFragPos;
out vec3 vViewPos;

// Constants
const float PI = 3.14159265359;
const int MAX_LIGHTS = 8;

// Struct definitions
struct Light {
    vec3 position;
    vec3 color;
    float intensity;
    float radius;
};

struct Material {
    vec3 ambient;
    vec3 diffuse;
    vec3 specular;
    float shininess;
};

// Uniform blocks
layout(std140) uniform LightBlock {
    Light lights[MAX_LIGHTS];
    int numLights;
};

uniform Material material;
uniform sampler2D uDiffuseMap;
uniform sampler2D uNormalMap;
uniform samplerCube uEnvironmentMap;

// Function declarations
vec3 calculateLighting(vec3 normal, vec3 fragPos, vec3 viewDir);
float calculateAttenuation(float distance, float radius);
vec3 fresnelSchlick(float cosTheta, vec3 F0);

void main() {
    // Transform position
    vec4 worldPos = uModel * vec4(aPosition, 1.0);
    vFragPos = worldPos.xyz;

    // Transform normal
    vNormal = normalize(uNormalMatrix * aNormal);

    // Pass through texture coordinates
    vTexCoord = aTexCoord;

    // Apply wave animation
    float wave = sin(aPosition.x * 2.0 + uTime) * 0.1;
    worldPos.y += wave;

    // Final position
    gl_Position = uProjection * uView * worldPos;
}

// Fragment shader typically in separate file, but showing functions:

vec3 calculateLighting(vec3 normal, vec3 fragPos, vec3 viewDir) {
    vec3 result = vec3(0.0);

    for (int i = 0; i < numLights && i < MAX_LIGHTS; i++) {
        Light light = lights[i];

        // Direction to light
        vec3 lightDir = normalize(light.position - fragPos);
        float distance = length(light.position - fragPos);

        // Diffuse
        float diff = max(dot(normal, lightDir), 0.0);
        vec3 diffuse = diff * light.color * material.diffuse;

        // Specular (Blinn-Phong)
        vec3 halfwayDir = normalize(lightDir + viewDir);
        float spec = pow(max(dot(normal, halfwayDir), 0.0), material.shininess);
        vec3 specular = spec * light.color * material.specular;

        // Attenuation
        float attenuation = calculateAttenuation(distance, light.radius);

        result += (diffuse + specular) * attenuation * light.intensity;
    }

    // Add ambient
    result += material.ambient * 0.1;

    return result;
}

float calculateAttenuation(float distance, float radius) {
    float attenuation = 1.0 / (1.0 + 0.09 * distance + 0.032 * distance * distance);
    float falloff = clamp(1.0 - distance / radius, 0.0, 1.0);
    return attenuation * falloff;
}

vec3 fresnelSchlick(float cosTheta, vec3 F0) {
    return F0 + (1.0 - F0) * pow(clamp(1.0 - cosTheta, 0.0, 1.0), 5.0);
}
