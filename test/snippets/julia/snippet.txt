# Recursive function
function factorial(n::Int)::Int
    n <= 1 ? 1 : n * factorial(n - 1)
end

# Tail-recursive with helper
function factorial_tail(n::Int)::Int
    function helper(n, acc)
        n <= 1 ? acc : helper(n - 1, n * acc)
    end
    helper(n, 1)
end

# Generators
function fibonacci()
    Channel{Int}(1) do channel
        a, b = 0, 1
        while true
            put!(channel, a)
            a, b = b, a + b
        end
    end
end

# Exception handling
function safe_divide(a::Number, b::Number)
    try
        a / b
    catch e
        if isa(e, DivideError)
            println("Cannot divide by zero")
            nothing
        else
            rethrow(e)
        end
    finally
        println("Division attempted")
    end
end
