# frozen_string_literal: true

# Module definition
module Greeting
  def greet
    "Hello, #{name}!"
  end
end

# Class definition
class Person
  include Greeting
  attr_accessor :name, :age
  attr_reader :id

  @@count = 0

  def initialize(name, age)
    @name = name
    @age = age
    @id = @@count += 1
  end

  def adult?
    age >= 18
  end

  def <=>(other)
    age <=> other.age
  end

  def to_s
    "Person(#{name}, #{age})"
  end

  class << self
    def create(name:, age:)
      new(name, age)
    end

    def count
      @@count
    end
  end

  private

  def secret_method
    "This is private"
  end
end

# Inheritance
class Employee < Person
  attr_accessor :role, :salary

  def initialize(name, age, role, salary)
    super(name, age)
    @role = role
    @salary = salary
  end

  def annual_salary
    salary * 12
  end
end

# Exception handling
def divide(a, b)
  raise ArgumentError, "Cannot divide by zero" if b.zero?
  a / b
rescue ArgumentError => e
  puts "Error: #{e.message}"
  nil
ensure
  puts "Division attempted"
end

# Enumerable
class Fibonacci
  include Enumerable

  def initialize(limit)
    @limit = limit
  end

  def each
    a, b = 0, 1
    while a <= @limit
      yield a
      a, b = b, a + b
    end
  end
end

# Module mixins
module Cacheable
  def cache
    @cache ||= {}
  end

  def cached(key, &block)
    cache[key] ||= block.call
  end
end

# Metaprogramming
class DynamicClass
  %w[foo bar baz].each do |method_name|
    define_method(method_name) do
      "Called #{method_name}"
    end
  end

  def method_missing(name, *args)
    "Unknown method: #{name}"
  end

  def respond_to_missing?(name, include_private = false)
    true
  end
end
