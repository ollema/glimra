# Struct (value type, immutable by default)
struct Point
  getter x : Float64
  getter y : Float64

  def initialize(@x, @y)
  end

  def distance_to(other : Point) : Float64
    Math.sqrt((other.x - @x) ** 2 + (other.y - @y) ** 2)
  end

  def +(other : Point) : Point
    Point.new(@x + other.x, @y + other.y)
  end
end

# Class
class Calculator
  property value : Int32

  def initialize(@value = 0)
  end

  def add(n : Int32) : self
    @value += n
    self
  end

  def subtract(n : Int32) : self
    @value -= n
    self
  end

  def self.create(initial : Int32) : Calculator
    new(initial)
  end
end

# Module as mixin
module Describable
  abstract def describe : String

  def print_description
    puts describe
  end
end

class Product
  include Describable
  include JSON::Serializable

  getter name : String
  getter price : Float64

  def initialize(@name, @price)
  end

  def describe : String
    "#{@name}: $#{@price}"
  end
end

# Generic class
class Container(T)
  getter items : Array(T)

  def initialize
    @items = [] of T
  end

  def add(item : T)
    @items << item
  end

  def first? : T?
    @items.first?
  end
end

# Recursive function
def factorial(n : Int32) : Int32
  return 1 if n <= 1
  n * factorial(n - 1)
end

# Higher-order function
def apply_twice(x : Int32, &block : Int32 -> Int32) : Int32
  block.call(block.call(x))
end

# Control flow
def classify(n : Int32) : String
  case n
  when .< 0
    "negative"
  when 0
    "zero"
  when 1..9
    "small"
  else
    "large"
  end
end

# Pattern matching with case
def describe_value(value)
  case value
  when Int32
    "integer: #{value}"
  when String
    "string: #{value}"
  when Array
    "array with #{value.size} elements"
  when {x: Int32, y: Int32}
    "point at (#{value[:x]}, #{value[:y]})"
  else
    "unknown"
  end
end

# Exception handling
def safe_divide(a : Int32, b : Int32) : Int32?
  begin
    a // b
  rescue DivisionByZeroError
    puts "Cannot divide by zero"
    nil
  ensure
    puts "Division attempted"
  end
end

# Raise with custom exception
class ValidationError < Exception
end

def validate_age(age : Int32) : Int32
  raise ValidationError.new("Age cannot be negative") if age < 0
  raise ValidationError.new("Age seems unrealistic") if age > 150
  age
end
